\section*{Overview}

This document gives the functional dependencies (FDs) of each relation in the \texttt{stock\_trading} database, and uses attribute closures to find every candidate key and count the superkeys. Relations and attributes match \texttt{sql/ddl.sql} and \texttt{docs/Schemata Overview/schemata.md}.

For a relation with attribute set $C$:
\begin{itemize}[nosep]
  \item $X$ is a \textbf{superkey} if $X^+ = C$.
  \item $X$ is a \textbf{candidate key} if it is a superkey and no proper subset of it is.
  \item \textbf{No-fire lemma:} if $X$ contains no FD's left-hand side, then $X^+ = X$. In particular $\emptyset^+ = \emptyset$.
\end{itemize}

\section{\texttt{user}}\label{r:user}

\[ C_U = \{\at{user\_id}, \at{name}, \at{email}, \at{cash\_balance}\} \]

\begin{itemize}[nosep]
  \fd{1}{\at{user\_id}}{\at{name}, \at{email}, \at{cash\_balance}}{Each user ID is one account.}
  \fd{2}{\at{email}}{\at{user\_id}}{An email registers one account (C.1.2).}
\end{itemize}

\textbf{Closures.} $\cl{\at{user\_id}} = C_U$ by \fdl{1}. $\cl{\at{email}} = C_U$ by \fdl{2} then \fdl{1}.

\textbf{Minimality.} Both keys are single attributes and $\emptyset^+ = \emptyset$.

\textbf{No others.} A set with neither \at{user\_id} nor \at{email} fires no FD, so every superkey contains one of them.

\textbf{Candidate keys:} $\{\at{user\_id}\}$ (primary), $\{\at{email}\}$. \textbf{Superkeys:} $2^3 + 2^3 - 2^2 = 12$.

% -----------------------------------------------------
\section{\texttt{portfolio}}\label{r:portfolio}

\[ C_P = \{\at{portfolio\_id}, \at{user\_id}, \at{portfolio\_name}, \at{created\_at}\} \]

\begin{itemize}[nosep]
  \fd{3}{\at{portfolio\_id}}{\at{user\_id}, \at{portfolio\_name}, \at{created\_at}}{Each portfolio ID is one account.}
  \fd{4}{\at{user\_id}, \at{portfolio\_name}}{\at{portfolio\_id}}{A user's portfolio names are unique (C.2.4).}
\end{itemize}

\textbf{Closures.} $\cl{\at{portfolio\_id}} = C_P$ by \fdl{3}. $\cl{\at{user\_id}, \at{portfolio\_name}} = C_P$ by \fdl{4} then \fdl{3}.

\textbf{Minimality.} $\cl{\at{user\_id}} = \{\at{user\_id}\}$ and $\cl{\at{portfolio\_name}} = \{\at{portfolio\_name}\}$ by the no-fire lemma, so neither half of the composite key is a superkey.

\textbf{No others.} A set without \at{portfolio\_id} and missing \at{user\_id} or \at{portfolio\_name} fires no FD.

\textbf{Candidate keys:} $\{\at{portfolio\_id}\}$ (primary), $\{\at{user\_id}, \at{portfolio\_name}\}$. \textbf{Superkeys:} $2^3 + 2^2 - 2^1 = 10$.

\section{\texttt{exchange}}\label{r:exchange}

\section{\texttt{stock\_price\_history}}\label{r:stock_price_history}

\[ C_H = \{\at{ticker}, \at{price\_date}, \at{opening\_price}, \at{closing\_price}, \at{high\_price}, \at{low\_price}, \at{volume}\} \]

\begin{itemize}[nosep]
  \fd{7}{\at{ticker}, \at{price\_date}}{\at{opening\_price}, \at{closing\_price}, \at{high\_price}, \at{low\_price}, \at{volume}}{One price record per stock per day (C.5.1).}
\end{itemize}

\textbf{Closure.} $\cl{\at{ticker}, \at{price\_date}} = C_H$ by \fdl{7}.

\textbf{Minimality.} $\cl{\at{ticker}} = \{\at{ticker}\}$ and $\cl{\at{price\_date}} = \{\at{price\_date}\}$ by the no-fire lemma.

\textbf{No others.} A set missing either attribute fires no FD.

\textbf{Candidate key:} $\{\at{ticker}, \at{price\_date}\}$. \textbf{Superkeys:} $2^5 = 32$.

\section{\texttt{watchlist}}\label{r:watchlist}

\[ C_W = \{\at{user\_id}, \at{ticker}, \at{target\_price}, \at{added\_at}\} \]

\begin{itemize}[nosep]
  \fd{8}{\at{user\_id}, \at{ticker}}{\at{target\_price}, \at{added\_at}}{A user watches a stock at most once (C.6.1).}
\end{itemize}

\textbf{Closure.} $\cl{\at{user\_id}, \at{ticker}} = C_W$ by \fdl{8}.

\textbf{Minimality.} $\cl{\at{user\_id}} = \{\at{user\_id}\}$ and $\cl{\at{ticker}} = \{\at{ticker}\}$ by the no-fire lemma. \fdl{1} belongs to \texttt{user}, and its right-hand side is not in $C_W$.

\textbf{No others.} A set missing either attribute fires no FD.

\textbf{Candidate key:} $\{\at{user\_id}, \at{ticker}\}$. \textbf{Superkeys:} $2^2 = 4$.
