\section*{Overview}

This document gives the relational algebra form of the \texttt{stock\_trading} constraints. Constraint IDs match \texttt{docs/Schemata Overview/schemata.md} and the comments in \texttt{sql/ddl.sql}. Each constraint is written in one of two forms:
\begin{itemize}[nosep]
  \item \textbf{Subset} ($\subseteq$): referential integrity, where every value in one relation must appear in another.
  \item \textbf{Empty set} ($= \emptyset$): domain rules, where selecting the violating tuples must return nothing.
\end{itemize}

Constraints with no meaningful relational algebra form are left out. These are: primary keys and \texttt{UNIQUE} (covered in the Candidate Keys document); \texttt{NOT NULL}; \texttt{DEFAULT}s; \texttt{ON DELETE} actions; and rules about how the database may change, which the triggers enforce rather than a property of a single state (C.7.6--C.7.9, C.8.6, C.8.8).

\section{\texttt{user}}

\begin{itemize}[nosep]
  \item \cl{1.4} Email has a valid format
  \item \cl{1.5} Cash balance is non-negative
\end{itemize}

\constraint{1.4}{Email has a valid format}
Contains \texttt{@} and a domain.
\[ \sigma_{\text{email} \text{ NOT LIKE } \text{'\%\_@\_\%.\_\%'}}(\text{User}) = \emptyset \]

\constraint{1.5}{Cash balance is non-negative}
A user can't spend money they don't have.
\[ \sigma_{\text{cash\_balance} < 0}(\text{User}) = \emptyset \]

% -----------------------------------------------------
\section{\texttt{portfolio}}

\begin{itemize}[nosep]
  \item \cl{2.2} Owner exists
  \item \cl{2.5} Name is non-empty
\end{itemize}

\constraint{2.2}{Owner exists}
Every portfolio belongs to a registered user.
\[ \pi_{\text{user\_id}}(\text{Portfolio}) \subseteq \pi_{\text{user\_id}}(\text{User}) \]

\constraint{2.5}{Name is non-empty}
\[ \sigma_{\text{portfolio\_name} = \text{''}}(\text{Portfolio}) = \emptyset \]

% -----------------------------------------------------
\section{\texttt{exchange}}

\begin{itemize}[nosep]
  \item \cl{3.3} Currency is a 3-letter code
  \item \cl{3.4} Exchange code is uppercase
\end{itemize}

\constraint{3.3}{Currency is a 3-letter code}
ISO 4217, e.g.\ \texttt{USD}, \texttt{CAD}.
\[ \sigma_{\text{CHAR\_LENGTH}(\text{currency}) \neq 3}(\text{Exchange}) = \emptyset \]

\constraint{3.4}{Exchange code is uppercase}
Compared case-sensitively.
\[ \sigma_{\text{exchange\_code} \neq \text{UPPER}(\text{exchange\_code})}(\text{Exchange}) = \emptyset \]

% -----------------------------------------------------
\section{\texttt{stock}}

\begin{itemize}[nosep]
  \item \cl{4.2} Exchange exists
  \item \cl{4.4} Ticker is uppercase
\end{itemize}

\constraint{4.2}{Exchange exists}
Every stock is listed on a known exchange.
\[ \pi_{\text{exchange\_code}}(\text{Stock}) \subseteq \pi_{\text{exchange\_code}}(\text{Exchange}) \]

\constraint{4.4}{Ticker is uppercase}
Compared case-sensitively.
\[ \sigma_{\text{ticker} \neq \text{UPPER}(\text{ticker})}(\text{Stock}) = \emptyset \]

% -----------------------------------------------------
\section{\texttt{stock\_price\_history}}

\begin{itemize}[nosep]
  \item \cl{5.2} Stock exists
  \item \cl{5.3} Prices are positive
  \item \cl{5.4} High is at least low
  \item \cl{5.5} Open and close lie within low--high
  \item \cl{5.6} Volume is non-negative
  \item \cl{5.7} Price date is not in the future
\end{itemize}

\constraint{5.2}{Stock exists}
A price record can't exist without its stock (weak entity).
\[ \pi_{\text{ticker}}(\text{StockPriceHistory}) \subseteq \pi_{\text{ticker}}(\text{Stock}) \]

\constraint{5.3}{Prices are positive}
\[ \sigma_{\text{opening\_price} \le 0 \,\lor\, \text{closing\_price} \le 0 \,\lor\, \text{high\_price} \le 0 \,\lor\, \text{low\_price} \le 0}(\text{StockPriceHistory}) = \emptyset \]

\constraint{5.4}{High is at least low}
\[ \sigma_{\text{high\_price} < \text{low\_price}}(\text{StockPriceHistory}) = \emptyset \]

\constraint{5.5}{Open and close lie within low--high}
\[ \sigma_{\text{opening\_price} < \text{low\_price} \,\lor\, \text{opening\_price} > \text{high\_price} \,\lor\, \text{closing\_price} < \text{low\_price} \,\lor\, \text{closing\_price} > \text{high\_price}}(\text{StockPriceHistory}) = \emptyset \]

\constraint{5.6}{Volume is non-negative}
\[ \sigma_{\text{volume} < 0}(\text{StockPriceHistory}) = \emptyset \]

\constraint{5.7}{Price date is not in the future}
Evaluated against the current date; enforced by trigger.
\[ \sigma_{\text{price\_date} > \text{CURDATE}()}(\text{StockPriceHistory}) = \emptyset \]

% -----------------------------------------------------
\section{\texttt{watchlist}}

\begin{itemize}[nosep]
  \item \cl{6.2} User exists
  \item \cl{6.3} Stock exists
  \item \cl{6.4} Target price, if set, is positive
\end{itemize}

\constraint{6.2}{User exists}
\[ \pi_{\text{user\_id}}(\text{Watchlist}) \subseteq \pi_{\text{user\_id}}(\text{User}) \]

\constraint{6.3}{Stock exists}
\[ \pi_{\text{ticker}}(\text{Watchlist}) \subseteq \pi_{\text{ticker}}(\text{Stock}) \]

\constraint{6.4}{Target price, if set, is positive}
\texttt{NULL} means no alert and is allowed.
\[ \sigma_{\text{target\_price} \neq \text{NULL} \,\land\, \text{target\_price} \le 0}(\text{Watchlist}) = \emptyset \]

% -----------------------------------------------------
\section{\texttt{transaction}}

\begin{itemize}[nosep]
  \item \cl{7.2} Portfolio and stock exist
  \item \cl{7.3} Quantity is positive
  \item \cl{7.4} Price is positive
  \item \cl{7.5} Type is buy or sell
  \item \cl{7.10} Execution time is not in the future
\end{itemize}

\constraint{7.2}{Portfolio and stock exist}
\begin{gather*}
  \pi_{\text{portfolio\_id}}(\text{Transaction}) \subseteq \pi_{\text{portfolio\_id}}(\text{Portfolio}) \\
  \pi_{\text{ticker}}(\text{Transaction}) \subseteq \pi_{\text{ticker}}(\text{Stock})
\end{gather*}

\constraint{7.3}{Quantity is positive}
\[ \sigma_{\text{quantity} \le 0}(\text{Transaction}) = \emptyset \]

\constraint{7.4}{Price is positive}
\[ \sigma_{\text{price} \le 0}(\text{Transaction}) = \emptyset \]

\constraint{7.5}{Type is buy or sell}
\[ \sigma_{\text{transaction\_type} \notin \{\text{'buy'}, \text{'sell'}\}}(\text{Transaction}) = \emptyset \]

\constraint{7.10}{Execution time is not in the future}
Evaluated against the current time; enforced by trigger.
\[ \sigma_{\text{executed\_at} > \text{NOW}()}(\text{Transaction}) = \emptyset \]

% -----------------------------------------------------
\section{\texttt{limit\_order}}

\begin{itemize}[nosep]
  \item \cl{8.2} Portfolio and stock exist
  \item \cl{8.3} Target price and quantity are positive
  \item \cl{8.4} Type is buy or sell
  \item \cl{8.7} Every executed order has a matching transaction
\end{itemize}

\constraint{8.2}{Portfolio and stock exist}
\begin{gather*}
  \pi_{\text{portfolio\_id}}(\text{LimitOrder}) \subseteq \pi_{\text{portfolio\_id}}(\text{Portfolio}) \\
  \pi_{\text{ticker}}(\text{LimitOrder}) \subseteq \pi_{\text{ticker}}(\text{Stock})
\end{gather*}

\constraint{8.3}{Target price and quantity are positive}
\[ \sigma_{\text{target\_price} \le 0 \,\lor\, \text{quantity} \le 0}(\text{LimitOrder}) = \emptyset \]

\constraint{8.4}{Type is buy or sell}
\[ \sigma_{\text{order\_type} \notin \{\text{'buy'}, \text{'sell'}\}}(\text{LimitOrder}) = \emptyset \]

\constraint{8.7}{Every executed order has a matching transaction}
The trade is recorded at the order's target price. Transaction attributes are renamed to match.
\begin{align*}
  &\pi_{\text{portfolio\_id},\, \text{ticker},\, \text{order\_type},\, \text{quantity},\, \text{target\_price}}\bigl(\sigma_{\text{status} = \text{'executed'}}(\text{LimitOrder})\bigr) \\
  &\quad\subseteq \rho_{(\text{portfolio\_id},\, \text{ticker},\, \text{order\_type},\, \text{quantity},\, \text{target\_price})}\bigl(\pi_{\text{portfolio\_id},\, \text{ticker},\, \text{transaction\_type},\, \text{quantity},\, \text{price}}(\text{Transaction})\bigr)
\end{align*}
